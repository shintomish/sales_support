## 抽出条件メモ（再現用 SQL）

```sql
-- 直近 30 日に汎用機/メインフレーム/COBOL 系メールを送ってきた engineer mail 担当者
WITH hot AS (
  SELECT unnest(ARRAY[
    't-techlife.com','sasatech.co.jp','arkanoids.jp','streetventure.biz',
    'code-d.co.jp','mail.ai-center.co.jp','iliostar.co.jp','n-ic.jp',
    'techlab-inc.co.jp','dx-incubation.com','1-r.co.jp','free-brain.co.jp',
    'careerbeat.jp','winvolve.co.jp'
  ]) AS bp_domain
)
SELECT
  split_part(e.from_address, '@', 2) AS bp_domain,
  e.from_name,
  e.from_address,
  COUNT(*) AS hot_mail_n,
  MAX(e.received_at) AS last_received,
  (ARRAY_AGG(e.subject ORDER BY e.received_at DESC))[1] AS latest_subject
FROM emails e
JOIN hot h ON h.bp_domain = split_part(e.from_address, '@', 2)
WHERE e.category = 'engineer'
  AND e.received_at > NOW() - INTERVAL '30 days'
  AND (
    e.body_text ILIKE '%汎用機%' OR e.body_text ILIKE '%メインフレーム%'
    OR e.subject ILIKE '%汎用機%' OR e.subject ILIKE '%メインフレーム%' OR e.subject ILIKE '%COBOL%'
  )
GROUP BY 1, 2, 3
ORDER BY 1, 4 DESC;
```
